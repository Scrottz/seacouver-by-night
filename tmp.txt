import Layout from '../layouts/Layout.astro';
import data from '../../../data/db.json';

const pcs = data.characters.filter(c => c.type === 'PC');
const npcs = data.characters.filter(c => c.type === 'NPC');

// Alle Bilder finden
const allImages = import.meta.glob('/public/res/img/**/*.{jpg,jpeg,png,webp}');

// Funktion: Findet alle Bilder für einen Charakter (z.B. aaliyah_khan_1, aaliyah_khan_2...)
const getGallery = (char) => {
    const slug = char.slug;
    const folder = char.type.toLowerCase();
    const pattern = `/public/res/img/${folder}/${slug}_`;

    return Object.keys(allImages)
        .filter(path => path.startsWith(pattern))
        .sort()
        .map(path => path.replace('/public', ''));
};

const groupedNpcs = npcs.reduce((acc, npc) => {
    const clan = npc.clan || "Unbekannt";
    if (!acc[clan]) acc[clan] = [];
    acc[clan].push({ ...npc, gallery: getGallery(npc) });
    return acc;
}, {} as Record<string, any[]>);

const sortedClans = Object.keys(groupedNpcs).sort();
const pcsWithGallery = pcs.map(pc => ({ ...pc, gallery: getGallery(pc) }));
